# 本机连接、监听与 TCP 信息；部分进程信息需要相应权限
ss -lnt
ss -tan
ss -tin

# 从当前网络命名空间看路由与邻居
ip route get 192.0.2.20
ip neigh

# 仅对一个已知 TCP 服务做连接检查，具体选项按本机 nc 版本确认
nc -vz -w 2 192.0.2.20 443

# 获授权后，在实际接口上限量观察一个方向相关的 TCP 报文
# -nn 避免名称解析造成额外干扰；不加 -A/-X 展示正文
# 抓包仍可能涉及敏感数据，应按批准的范围与留存规则处理
 tcpdump -nn -i IFACE -c 100 'host 192.0.2.20 and tcp port 443'
